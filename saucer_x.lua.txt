local function say(text)

		StarterGui:SetCore("SendNotification", { Title = "Saucer-X", Text = text, Duration = 3 })

-- Анимации групп (двери / капот / багажник / шасси)
---------------------------------------------------------------------
local function reg(group, weld, closed, open)
	table.insert(anim[group], { w = weld, c = closed, o = open })
end

local function setGroup(group, open, instant)
	state[group] = open
	for _, a in ipairs(anim[group]) do
		if a.w and a.w.Parent then
			local target = open and a.o or a.c
			if instant then
				a.w.C0 = target
			else
				TweenService:Create(a.w, TweenInfo.new(0.9, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), { C0 = target }):Play()
			end
		end
	end
end

local BLOCK, BALL, CYL = Enum.PartType.Block, Enum.PartType.Ball, Enum.PartType.Cylinder

local function mk(name, shape, size, rel, color, mat, o)
	o = o or {}
	local p = Instance.new("Part")
	p.Name = name
	p.Shape = shape
	p.Size = size
	p.Color = color
	p.Material = mat or Enum.Material.Metal
	p.Anchored = false
	p.CanCollide = o.collide or false
	p.CanTouch = false
	p.Massless = true
	p.Transparency = o.tr or 0
	p.Reflectance = o.refl or 0
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.CFrame = root.CFrame * rel
	p.Parent = ship
	local w = Instance.new("Weld")
	w.Part0 = root
	w.Part1 = p
	w.C0 = rel
	w.Parent = p
	return p, w
end

local function disc(name, y, h, d, color, mat, o)
	return mk(name, CYL, Vector3.new(h, d, d), CFrame.new(0, y, 0) * CFrame.Angles(0, 0, math.pi / 2), color, mat, o)
end

local function zcyl(name, len, d, rel, color, mat, o)
	return mk(name, CYL, Vector3.new(len, d, d), rel * CFrame.Angles(0, math.pi / 2, 0), color, mat, o)
end

local function prompt(rel, action, cb, dist)
	local p = mk("Prompt", BLOCK, Vector3.new(1, 1, 1), rel, C.dark, nil, { tr = 1 })
	local pp = Instance.new("ProximityPrompt")
	pp.ActionText = action
	pp.ObjectText = "Saucer-X"
	pp.HoldDuration = 0
	pp.RequiresLineOfSight = false
	pp.MaxActivationDistance = dist or 10
	pp.Parent = p
	pp.Triggered:Connect(cb)
	table.insert(promptList, pp)
end

local function setPrompts(on)
	for _, pp in ipairs(promptList) do
		if pp and pp.Parent then pp.Enabled = on end
	end
end

-- крышка на петле (капот dir=-1, багажник dir=+1)
local function compartment(group, dir, wallH, ang, fill)
	local zc = dir * 7.05
	local base = 1.8
	local top = base + wallH
	mk(group .. "WL", BLOCK, Vector3.new(0.2, wallH, 2.1), CFrame.new(-2.1, base + wallH / 2, zc), C.hull2)
	mk(group .. "WR", BLOCK, Vector3.new(0.2, wallH, 2.1), CFrame.new(2.1, base + wallH / 2, zc), C.hull2)
	mk(group .. "WF", BLOCK, Vector3.new(4.4, wallH, 0.2), CFrame.new(0, base + wallH / 2, dir * 8.1), C.hull2)
	mk(group .. "Floor", BLOCK, Vector3.new(4, 0.1, 2.1), CFrame.new(0, 1.85, zc), C.black)
	fill(zc)

	local closed = CFrame.new(0, top + 0.1, zc)
	local hinge = Vector3.new(0, top + 0.1, dir * 6.0)
	local open = CFrame.new(hinge) * CFrame.Angles(-dir * math.rad(ang), 0, 0) * CFrame.new(-hinge) * closed
	local delta = open * closed:Inverse()

	local _, w = mk(group .. "Lid", BLOCK, Vector3.new(4.4, 0.2, 2.2), closed, C.hull, Enum.Material.Metal)
	reg(group, w, closed, open)

	local sRel = CFrame.new(0, top + 0.22, zc)
	local _, w2 = mk(group .. "LidStripe", BLOCK, Vector3.new(0.3, 0.05, 1.7), sRel, C.orange, Enum.Material.Neon)
	reg(group, w2, sRel, delta * sRel)

	local hRel = CFrame.new(0, top + 0.22, zc + dir * 0.9)
	local _, w3 = mk(group .. "LidHandle", BLOCK, Vector3.new(1.0, 0.08, 0.2), hRel, C.dark)
	reg(group, w3, hRel, delta * hRel)

	for _, sx in ipairs({ -1.6, 1.6 }) do
		local bRel = CFrame.new(sx, top + 0.23, zc - dir * 0.8)
		local _, wb = mk(group .. "LidBolt", CYL, Vector3.new(0.06, 0.28, 0.28), bRel * CFrame.Angles(0, 0, math.pi / 2), C.dark)
		reg(group, wb, bRel * CFrame.Angles(0, 0, math.pi / 2), delta * bRel * CFrame.Angles(0, 0, math.pi / 2))
	end
end

local function build(cf)

	root.Name = "Root"
	root.Size = Vector3.new(1, 1, 1)
	root.Transparency = 1
	root.Anchored = true
	root.CanCollide = false
	root.CanTouch = false
	root.CanQuery = false
	root.CFrame = cf
	root.Parent = ship
	ship.PrimaryPart = root

	lensList, lightList, promptList = {}, {}, {}
	anim = { doors = {}, hood = {}, trunk = {}, wheels = {} }

	disc("Belly2", -0.35, 0.5, 11.5, C.hull2, Enum.Material.Metal)

	setLights(state.lights)
end

function setLights(on)
	state.lights = on
	for _, l in ipairs(lensList) do
		if l and l.Parent then
			l.Material = on and Enum.Material.Neon or Enum.Material.SmoothPlastic
		end
	end
	for _, l in ipairs(lightList) do
		if l and l.Parent then l.Enabled = on end
	end
end

---------------------------------------------------------------------
-- Спавн / удаление
---------------------------------------------------------------------
local function destroyShip()
	if state.seated then exitShip() end
	if ship then ship:Destroy() end
	ship, root, beacon, thrustEmitter = nil, nil, nil, nil
end

local function spawnShip()
	destroyShip()
	local char = lp.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then say("Персонаж не найден") return end
	local look = hrp.CFrame.LookVector
	look = Vector3.new(look.X, 0, look.Z)
	if look.Magnitude < 0.01 then look = Vector3.new(0, 0, -1) end
	look = look.Unit
	local p = hrp.Position + look * 24
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { char }
	local hit = Workspace:Raycast(p + Vector3.new(0, 60, 0), Vector3.new(0, -300, 0), rp)
	local y = hit and (hit.Position.Y + 2.1) or (hrp.Position.Y - 1.0)
	yaw = math.atan2(-look.X, -look.Z) + math.pi
	pitch, roll, vel = 0, 0, Vector3.zero
	pos = Vector3.new(p.X, y, p.Z)
	build(CFrame.new(pos) * CFrame.Angles(0, yaw, 0))
	updateSeatBtn()
	say("Тарелка готова. Подойди и нажми «Сесть» (G)")
end

---------------------------------------------------------------------
-- Посадка / выход
---------------------------------------------------------------------
function updateSeatBtn()
	if seatBtn then seatBtn.Text = state.seated and "Выйти (G)" or "Сесть (G)" end
end

function enterShip()
	if state.seated or busy then return end
	if not ship then say("Сначала создай тарелку") return end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) then return end
	if (hrp.Position - root.Position).Magnitude > 45 then
		say("Подойди ближе к тарелке (до 45 стадов)")
		return
	end
	busy = true
	setGroup("doors", true)
	task.spawn(function()
		task.wait(0.7)
		if ship and ship.Parent and hum.Parent then
			yaw = select(2, root.CFrame:ToEulerAnglesYXZ())
			pitch, roll, vel = 0, 0, Vector3.zero
			pos = root.Position
			state.seated = true
			hum.AutoRotate = false
			hrp.Anchored = true
			hrp.CFrame = root.CFrame * seatRel
			hum.Sit = true

			flightParams = RaycastParams.new()
			flightParams.FilterType = Enum.RaycastFilterType.Exclude
			flightParams.FilterDescendantsInstances = { ship, char }

			local cam = Workspace.CurrentCamera
			saved.fov = cam and cam.FieldOfView or 70
			saved.min, saved.max = lp.CameraMinZoomDistance, lp.CameraMaxZoomDistance
			lp.CameraMaxZoomDistance = 150
			lp.CameraMinZoomDistance = 28

			if fly then fly.Visible = true end
			if hud then hud.Visible = true end
			setPrompts(false)
			updateSeatBtn()
			task.wait(0.6)
			setGroup("doors", false)
			say("Полёт: WASD / джойстик, Space или ▲ — вверх, Shift — турбо")
		end
		busy = false
	end)
end

function exitShip()
	if not state.seated then return end
	state.seated = false
	if fly then fly.Visible = false end
	if hud then hud.Visible = false end
	if thrustEmitter then thrustEmitter.Enabled = false end
	setPrompts(true)

	local cam = Workspace.CurrentCamera
	if cam and saved.fov then cam.FieldOfView = saved.fov end
	if saved.min then lp.CameraMinZoomDistance = saved.min end
	if saved.max then lp.CameraMaxZoomDistance = saved.max end

	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp and root then
		hrp.Anchored = false
		local out = root.CFrame * CFrame.new(11.5, 0, 0)
		local rp = RaycastParams.new()
		rp.FilterType = Enum.RaycastFilterType.Exclude
		rp.FilterDescendantsInstances = { ship, char }
		local hit = Workspace:Raycast(out.Position + Vector3.new(0, 6, 0), Vector3.new(0, -60, 0), rp)
		local y = hit and (hit.Position.Y + 3.2) or out.Position.Y
		hrp.CFrame = CFrame.new(out.X, y, out.Z)
		hrp.AssemblyLinearVelocity = Vector3.zero
	elseif hrp then
		hrp.Anchored = false
	end
	if hum then
		hum.Sit = false
		hum.AutoRotate = true
		hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
	updateSeatBtn()
	if ship then
		setGroup("doors", true)
		task.delay(2, function()
			if ship and ship.Parent and not state.seated then setGroup("doors", false) end
		end)
	end
end

local controls
pcall(function()
	local ps = lp:FindFirstChild("PlayerScripts")
	local pm = ps and ps:FindFirstChild("PlayerModule")
	if pm then controls = require(pm):GetControls() end
end)

local function getMove()
	local mv = Vector3.zero
	if controls then
		local ok, v = pcall(function() return controls:GetMoveVector() end)
		if ok and v then mv = v end
	end
	if mv.Magnitude < 0.05 then
		local x, z = 0, 0
		if UIS:IsKeyDown(Enum.KeyCode.W) then z = z - 1 end
		if UIS:IsKeyDown(Enum.KeyCode.S) then z = z + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.D) then x = x + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.A) then x = x - 1 end
		mv = Vector3.new(x, 0, z)
	end
	return mv
end

local function sweep(p, step)
	if step.Magnitude < 0.01 or not flightParams then return nil end
	local dir = step.Unit * (step.Magnitude + 2)
	local origin = p + Vector3.new(0, 3.5, 0)
	local ok, res = pcall(function()
		return Workspace:Spherecast(origin, 5, dir, flightParams)
	end)
	if not ok then
		res = Workspace:Raycast(origin, dir + step.Unit * 4, flightParams)
	end
	return res
end

table.insert(connections, RunService.RenderStepped:Connect(function(dt)
	if not (ship and ship.Parent and root) then return end
	dt = math.min(dt, 0.1)
	local t = os.clock()
	if beacon then beacon.Transparency = (math.sin(t * 5) > 0) and 0 or 0.7 end
	if not state.seated then return end

	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) or hum.Health <= 0 then exitShip() return end

	local mv = getMove()
	local fwd = -mv.Z
	local turn = mv.X
	local boost = (UIS:IsKeyDown(Enum.KeyCode.LeftShift) or turbo) and 2.2 or 1
	local up = 0
	if UIS:IsKeyDown(Enum.KeyCode.Space) or vertUp then up = up + 1 end
	if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.Q) or vertDown then up = up - 1 end

	yaw = yaw - turn * dt * 1.8
	local heading = CFrame.Angles(0, yaw, 0)
	local target = heading.LookVector * fwd * speed * boost + Vector3.new(0, up * speed * 0.6, 0)
	vel = vel:Lerp(target, math.clamp(dt * 2.5, 0, 1))

	-- столкновения со стенами
	local step = vel * dt
	local res = sweep(pos, step)
	if res and res.Distance < step.Magnitude + 1.2 then
		local n = res.Normal
		local d = vel:Dot(n)
		if d < 0 then vel = vel - n * d end
		step = vel * dt
	end
	pos = pos + step

	-- земля и шасси
	local hit = Workspace:Raycast(pos + Vector3.new(0, 3, 0), Vector3.new(0, -9, 0), flightParams)
	if hit then
		local minY = hit.Position.Y + 2.1
		if pos.Y < minY then
			pos = Vector3.new(pos.X, minY, pos.Z)
			if vel.Y < 0 then vel = Vector3.new(vel.X, 0, vel.Z) end
		end
	end
	local wantUp = (not hit) or (pos.Y - hit.Position.Y) > 6
	if wantUp ~= state.wheels then setGroup("wheels", wantUp) end

	-- наклоны корпуса
	local tp = -math.clamp(fwd, -1, 1) * 0.14
	local tr = -math.clamp(turn, -1, 1) * 0.3
	local k = math.min(dt * 4, 1)
	pitch = pitch + (tp - pitch) * k
	roll = roll + (tr - roll) * k

	local bob = math.sin(t * 2) * 0.12
	root.CFrame = CFrame.new(pos + Vector3.new(0, bob, 0)) * heading * CFrame.Angles(pitch, 0, roll)

	hrp.CFrame = root.CFrame * seatRel
	hrp.AssemblyLinearVelocity = Vector3.zero
	if not hum.Sit then hum.Sit = true end

	-- эффекты и HUD
	local spd = vel.Magnitude
	local frac = math.clamp(spd / (speed * 2), 0, 1)
	if thrustEmitter then
		thrustEmitter.Enabled = true
		thrustEmitter.Rate = 25 + frac * 90
	end
	local cam = Workspace.CurrentCamera
	if cam and saved.fov then
		cam.FieldOfView = cam.FieldOfView + ((saved.fov + frac * 18) - cam.FieldOfView) * math.min(dt * 3, 1)
	end
	if hud then
		hud.Text = string.format("СКОРОСТЬ %d    ВЫСОТА %d", spd, pos.Y)
	end
end))
