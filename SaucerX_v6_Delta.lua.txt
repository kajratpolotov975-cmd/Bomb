local function hinge(group, weld, closed, H, ang, part, collFn)
	local Hi = H:Inverse()
	table.insert(anim[group], {
		w = weld,
		part = part,
		coll = collFn,
		fn = function(a) return H * CFrame.Angles(ang * a, 0, 0) * Hi * closed end,
	})
end

local function hingeDoor(group, weld, closed, H, openYaw, openPitch, part, collFn)
	local Hi = H:Inverse()
	table.insert(anim[group], {
		w = weld,
		part = part,
		coll = collFn,
		fn = function(a)
			return H * CFrame.Angles(0, openYaw * a, 0) * CFrame.Angles(openPitch * a, 0, 0) * Hi * closed
		end,
	})
end

local function applyAlpha(group, a)
	if group == "gear" then
		updateWheels()
		return
	end
	for _, e in ipairs(anim[group]) do
		if e.w and e.w.Parent then
			e.w.C0 = e.fn(a)
			if e.part and e.coll then e.part.CanCollide = e.coll(a) end
		end
	end
end

local function setGroup(group, open, instant)
	state[group] = open
	gen[group] = gen[group] + 1
	local my = gen[group]
	local from = alpha[group]
	local to = open and 1 or 0
	if instant or from == to then
		alpha[group] = to
		applyAlpha(group, to)
	else
		task.spawn(function()
			local t0 = os.clock()
			local dur = (group == "gear" and 0.85 or 0.95) * math.abs(to - from)
			while gen[group] == my do
				local p = math.min((os.clock() - t0) / dur, 1)
				local e = (p < 0.5) and (2 * p * p) or (1 - 2 * (1 - p) * (1 - p))
				alpha[group] = from + (to - from) * e
				applyAlpha(group, alpha[group])
				if p >= 1 then break end
				RunService.Heartbeat:Wait()
			end
		end)
	end
	if refreshUI then refreshUI() end
end

local function triggerAction(action)
	if not ship then return end
	local now = os.clock()
	if now - lastClickTime < 0.22 then return end
	lastClickTime = now

	if action == "doorL" then
		setGroup("doorL", not state.doorL)
		say(state.doorL and "Левая дверь открыта — заходи по трапу на кресло!" or "Левая дверь закрыта")
	elseif action == "doorR" then
		setGroup("doorR", not state.doorR)
		say(state.doorR and "Правая дверь открыта — заходи внутрь!" or "Правая дверь закрыта")
	elseif action == "hood" then
		setGroup("hood", not state.hood)
		say(state.hood and "Капот открыт: реактор, вентилятор и катушка Теслы" or "Капот закрыт")
	elseif action == "trunk" then
		setGroup("trunk", not state.trunk)
		say(state.trunk and "Багажник открыт: портальная пушка и Meeseeks-box" or "Багажник закрыт")
	elseif action == "lights" then
		setLights(not state.lights)
		say(state.lights and "Фары включены (светят на дорогу)" or "Фары выключены")
	elseif action == "seatL" or action == "seatR" then
		local side = (action == "seatR") and 1 or -1
		if state.seated then
			if state.seatSide ~= side then
				state.seatSide = side
				say(side < 0 and "Пересел за руль!" or "Пересел на пассажирское кресло!")
			elseif vel.Magnitude < 2 then
				exitShip()
			else
				say("Чтобы встать с кресла, нажми Прыжок (Space) или G")
			end
		else
			local char = lp.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hum and hrp and root then
				local seatCF = root.CFrame * (side < 0 and seatRelL or seatRelR)
				if (hrp.Position - seatCF.Position).Magnitude <= 15 then
					sitDown(hum, hrp, side)
				else
					say("Открой дверь нажатием, зайди по трапу и подойди к креслу!")
				end
			end
		end
	end
end

local function makeClickable(part, action, dist)
	if not part then return end
	part.CanQuery = true
	partActions[part] = action
	table.insert(clickableParts, part)
	local cd = Instance.new("ClickDetector")
	cd.MaxActivationDistance = dist or 42
	cd.Parent = part
	cd.MouseClick:Connect(function(plr)
		if plr == lp then triggerAction(action) end
	end)
end

local BLOCK, BALL, CYL = Enum.PartType.Block, Enum.PartType.Ball, Enum.PartType.Cylinder
local MAT = Enum.Material

local N       = 16
local STEP    = 360 / N
local TANH    = math.tan(math.rad(STEP / 2))
local Y0, Y1, Y2 = 0.1, 0.95, 2.55       -- высота борта
local R0, R2  = 10.6, 7.6                -- радиус низа 10.6 (диаметр 21.2), верха 7.6
local DR, DH  = 6.0, 5.65                -- ФИЗИЧЕСКИЙ КУПОЛ: радиус 6.0, высота 5.65 (проём двери 6.7 стада!)

local function rOf(y) return R0 - (y - Y0) * (R0 - R2) / (Y2 - Y0) end
local function thetaOf(j) return math.rad(STEP / 2 + STEP * j) end
local function wallW(r) return 2 * r * TANH + 0.08 end

local function slabCF(theta, ya, ra, yb, rb)
	local dy, dr = yb - ya, ra - rb
	local len = math.sqrt(dy * dy + dr * dr)
	local a = math.atan2(dr, dy)
	local cf = CFrame.Angles(0, theta, 0) * CFrame.new(0, (ya + yb) / 2, -(ra + rb) / 2) * CFrame.Angles(a, 0, 0)
	return cf, len
end

local function mk(name, shape, size, rel, color, mat, o)
	o = o or {}
	local p = Instance.new("Part")
	p.Name = name
	p.Shape = shape
	p.Size = size
	p.Color = color
	p.Material = mat or MAT.Metal
	p.Anchored = false
	p.CanCollide = o.collide or false
	p.CanTouch = o.touch or false
	p.CanQuery = o.query ~= false
	p.Massless = true
	p.Transparency = o.tr or 0
	p.Reflectance = o.refl or 0
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.CFrame = root.CFrame * rel
	p.Parent = ship
	local w = Instance.new("Weld")
	w.Part0 = root
	w.Part1 = p
	w.C0 = rel
	w.Parent = p
	if o.action then makeClickable(p, o.action, o.clickDist) end
	return p, w
end

local function disc(name, y, h, d, color, mat, o)
	return mk(name, CYL, Vector3.new(h, d, d), CFrame.new(0, y, 0) * CFrame.Angles(0, 0, math.pi / 2), color, mat, o)
end

local function zcyl(name, len, d, rel, color, mat, o)
	return mk(name, CYL, Vector3.new(len, d, d), rel * CFrame.Angles(0, math.pi / 2, 0), color, mat, o)
end

local function vcyl(name, h, d, rel, color, mat, o)
	return mk(name, CYL, Vector3.new(h, d, d), rel * CFrame.Angles(0, 0, math.pi / 2), color, mat, o)
end

local function addLight(part, color, brightness, range)
	local l = Instance.new("PointLight")
	l.Color = color; l.Brightness = brightness; l.Range = range
	l.Parent = part
	return l
end

local function rotAxis(axis, a)
	if axis == "X" then return CFrame.Angles(a, 0, 0) end
	if axis == "Y" then return CFrame.Angles(0, a, 0) end
	return CFrame.Angles(0, 0, a)
end

local function spinPart(name, shape, size, frame, off, color, mat, axis, getter, o)
	local p, w = mk(name, shape, size, frame * off, color, mat, o)
	table.insert(spinners, { w = w, cf = frame, off = off, axis = axis, get = getter })
	return p
end

local function addPulse(part, light, base, kind)
	table.insert(pulsers, { part = part, light = light, base = base, kind = kind or 0 })
end

local function updateSpinners()
	for _, e in ipairs(spinners) do
		if e.w and e.w.Parent then
			e.w.C0 = e.cf * rotAxis(e.axis, e.get()) * e.off
		end
	end
end

local function updatePulses(t)
	for i, e in ipairs(pulsers) do
		local k = 0.5 + 0.5 * math.sin(t * 3 + i * 1.7)
		if e.part and e.part.Parent then e.part.Transparency = 0.05 + 0.25 * (1 - k) end
		if e.light and e.light.Parent then e.light.Brightness = e.base * (0.6 + 0.6 * k) end
	end
end

local function updatePodFlames(t, dt, active, boostMult, speedRatio)
	local target = active and math.clamp(0.55 + speedRatio * 0.45 + (boostMult > 1 and 0.35 or 0), 0.55, 1.35) or 0
	flamePower = flamePower + (target - flamePower) * math.min(dt * 10, 1)
	local show = flamePower > 0.03

	for idx, f in ipairs(podFlames) do
		if f.emitter and f.emitter.Parent then
			f.emitter.Enabled = show
			if show then
				f.emitter.Rate = math.floor(65 * flamePower)
				local spd = 16 + 22 * flamePower
				f.emitter.Speed = NumberRange.new(spd * 0.75, spd * 1.15)
			end
		end

		if not show then
			if f.outer and f.outer.Parent then f.outer.Transparency = 1 end
			if f.mid and f.mid.Parent then f.mid.Transparency = 1 end
			if f.core and f.core.Parent then f.core.Transparency = 1 end
			if f.tip and f.tip.Parent then f.tip.Transparency = 1 end
			if f.ring1 and f.ring1.Parent then f.ring1.Transparency = 1 end
			if f.ring2 and f.ring2.Parent then f.ring2.Transparency = 1 end
			if f.light and f.light.Parent then f.light.Brightness = 1.8 end
		else
			local flick1 = 0.88 + 0.22 * math.sin(t * 42 + idx * 2.1) + 0.1 * math.cos(t * 73 + idx)
			local flick2 = 0.90 + 0.18 * math.cos(t * 55 + idx * 1.4)
			local lenOuter = (3.4 + 2.8 * (flamePower - 0.5)) * flick1
			local lenMid   = lenOuter * 0.76 * flick2
			local lenCore  = lenOuter * 0.48

			if f.outer and f.outerW and f.outer.Parent then
				f.outer.Size = Vector3.new(lenOuter, 2.05 * flick2, 2.05 * flick2)
				f.outerW.C0 = f.baseCF * CFrame.new(0, 0, 2.32 + lenOuter / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.outer.Transparency = math.clamp(0.32 - 0.12 * flamePower + 0.08 * math.sin(t * 38), 0.15, 0.55)
			end
			if f.mid and f.midW and f.mid.Parent then
				f.mid.Size = Vector3.new(lenMid, 1.42 * flick1, 1.42 * flick1)
				f.midW.C0 = f.baseCF * CFrame.new(0, 0, 2.30 + lenMid / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.mid.Transparency = math.clamp(0.18 + 0.08 * math.cos(t * 48), 0.1, 0.35)
			end
			if f.core and f.coreW and f.core.Parent then
				f.core.Size = Vector3.new(lenCore, 0.86, 0.86)
				f.coreW.C0 = f.baseCF * CFrame.new(0, 0, 2.28 + lenCore / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.core.Transparency = 0.05
			end
			if f.tip and f.tipW and f.tip.Parent then
				local td = 1.25 * flick1
				f.tip.Size = Vector3.new(td, td, td * 1.4)
				f.tipW.C0 = f.baseCF * CFrame.new(0, 0, 2.32 + lenOuter)
				f.tip.Transparency = 0.38
			end

			local p1 = (t * 4.5 + idx * 0.3) % 1
			local p2 = (t * 4.5 + idx * 0.3 + 0.5) % 1
			if f.ring1 and f.ring1W and f.ring1.Parent then
				local d1 = 2.3 * (1 - p1 * 0.55)
				f.ring1.Size = Vector3.new(0.14, d1, d1)
				f.ring1W.C0 = f.baseCF * CFrame.new(0, 0, 2.35 + p1 * lenOuter * 0.95) * CFrame.Angles(0, math.pi / 2, 0)
				f.ring1.Transparency = 0.15 + p1 * 0.75
			end
			if f.ring2 and f.ring2W and f.ring2.Parent then
				local d2 = 2.3 * (1 - p2 * 0.55)
				f.ring2.Size = Vector3.new(0.14, d2, d2)
				f.ring2W.C0 = f.baseCF * CFrame.new(0, 0, 2.35 + p2 * lenOuter * 0.95) * CFrame.Angles(0, math.pi / 2, 0)
				f.ring2.Transparency = 0.15 + p2 * 0.75
			end

			if f.light and f.light.Parent then
				f.light.Brightness = 2.6 + 2.2 * flamePower * flick1
				f.light.Range = 22 + 10 * flamePower
			end
		end
	end
end

-- крышка: капот (dir = -1, спереди) или багажник (dir = +1, сзади). Открывается кликом прямо по ней!
local function compartment(group, dir)
	local axisTheta = (dir < 0) and 0 or math.pi
	local js = (dir < 0) and { 15, 0 } or { 7, 8 }
	local H = CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2, -HINGE_R)
	local LID_ANGLE = math.rad(92)

	local function add(name, shape, size, rel, color, mat, o)
		o = o or {}
		o.action = group
		local p, w = mk(name, shape, size, rel, color, mat, o)
		local collFn = (o and o.collide) and function(a) return a < 0.1 end or nil
		hinge(group, w, rel, H, LID_ANGLE, p, collFn)
		return p
	end

	local flatMid = (HINGE_R + R2) / 2
	for idx, j in ipairs(js) do
		local th = thetaOf(j)
		add(group .. "LidTop", BLOCK, Vector3.new(wallW(R2), 0.3, R2 - HINGE_R),
			CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.15 + (idx - 1) * 0.012, -flatMid), C.hull, MAT.Metal, { collide = true })
		local scf, slen = slabCF(th, Y1, rOf(Y1), Y2, R2)
		add(group .. "LidSlope", BLOCK, Vector3.new(wallW(rOf(Y1)), slen, 0.3), scf, C.hull, MAT.Metal, { collide = true })
	end

	add(group .. "LidStripe", BLOCK, Vector3.new(0.32, 0.05, R2 - HINGE_R - 0.25),
		CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2 + 0.03, -flatMid), C.orange, MAT.Neon)
	local lcf, llen = slabCF(axisTheta, Y1, rOf(Y1), Y2, R2)
	add(group .. "LidSlopeStripe", BLOCK, Vector3.new(0.32, llen, 0.06), lcf * CFrame.new(0, 0, -0.17), C.orange, MAT.Neon)
	for i = 1, 5 do
		add(group .. "LidLED", BLOCK, Vector3.new(0.38, 0.05, 0.08), lcf * CFrame.new(0, -llen / 2 + (i - 0.5) * llen / 5, -0.2), C.ledDark, MAT.Metal)
	end
	local wcf, wlen = slabCF(axisTheta, 0.25, rOf(0.25), Y1, rOf(Y1))
	mk(group .. "BodyStripe", BLOCK, Vector3.new(0.32, wlen, 0.06), wcf * CFrame.new(0, 0, -0.17), C.orange, MAT.Neon, { action = group })
	for i = 1, 4 do
		mk(group .. "BodyLED", BLOCK, Vector3.new(0.38, 0.05, 0.08), wcf * CFrame.new(0, -wlen / 2 + (i - 0.5) * wlen / 4, -0.2), C.ledDark, MAT.Metal, { action = group })
	end

	local hcf = slabCF(axisTheta, 1.1, rOf(1.1), 1.5, rOf(1.5)) * CFrame.new(0, 0, -0.2)
	add(group .. "LidHandle", BLOCK, Vector3.new(1.75, 0.22, 0.22), hcf, C.dark, MAT.Metal)

	if dir < 0 then
		local plateCF = CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2 + 0.12, -(HINGE_R + 0.75))
		add("LidPlate", BLOCK, Vector3.new(5.4, 0.36, 1.4), plateCF, C.hull, MAT.Metal)
		for _, sx in ipairs({ -1.5, 1.5 }) do
			local sc = plateCF * CFrame.new(sx, 0.02, -0.68)
			add("Screw", CYL, Vector3.new(0.12, 0.44, 0.44), sc * CFrame.Angles(0, math.pi / 2, 0), C.hull2, MAT.Metal)
			add("ScrewSlot", BLOCK, Vector3.new(0.04, 0.34, 0.06), sc * CFrame.new(0, 0, -0.05) * CFrame.Angles(0, 0, 0.25), C.dark, MAT.Metal)
		end
	end

	mk(group .. "Firewall", BLOCK, Vector3.new(4.2, 1.8, 0.2), CFrame.new(0, 1.4, dir * 5.4), C.hull2, MAT.Metal, { collide = true })
	local L = CFrame.new(0, 0.5, dir * 7.0) * ((dir > 0) and CFrame.Angles(0, math.pi, 0) or CFrame.new())
	if dir < 0 then hoodBay(L) else trunkBay(L) end
end

local function buildWheels()
	local tireCol = Color3.fromRGB(28, 30, 34)
	for _, z in ipairs({ -4.3, 4.3 }) do
		for _, s in ipairs({ -1, 1 }) do
			local out = CFrame.new(s * 6.3, -1.0, z)
			local inn = CFrame.new(s * 4.3, 0.15, z) * CFrame.Angles(0, 0, s * 1.35)
			local wh = { out = out, inn = inn, parts = {} }
			local function part(name, shape, size, off, color, mat, spin)
				local p, w = mk(name, shape, size, out * off, color, mat, { query = false })
				table.insert(wh.parts, { w = w, off = off, spin = spin })
				return p
			end
			local face = s * 0.44
			part("Tire", CYL, Vector3.new(0.95, 2.2, 2.2), CFrame.new(), tireCol, MAT.SmoothPlastic, true)
			part("Rim", CYL, Vector3.new(0.98, 1.3, 1.3), CFrame.new(), C.hull, MAT.Metal, true)
			part("Cap", CYL, Vector3.new(1.04, 0.5, 0.5), CFrame.new(), C.brass, MAT.Metal, true)
			part("Spoke1", BLOCK, Vector3.new(0.1, 1.15, 0.22), CFrame.new(face, 0, 0), C.dark, MAT.Metal, true)
			part("Spoke2", BLOCK, Vector3.new(0.1, 0.22, 1.15), CFrame.new(face, 0, 0), C.dark, MAT.Metal, true)
			part("Strut", CYL, Vector3.new(1.3, 0.18, 0.18), CFrame.new(face * 0.4, 0.65, 0) * CFrame.Angles(0, 0, math.pi / 2), C.dark, MAT.Metal, false)
			part("Piston", CYL, Vector3.new(1.2, 0.25, 0.25), CFrame.new(face * 0.4, 1.2, 0) * CFrame.Angles(0, 0, math.pi / 2), C.silver, MAT.Metal, false)
			table.insert(wheels, wh)
		end
	end
end

function updateWheels()
	local a = alpha.gear
	if a == lastWheelA and wheelSpin == lastWheelSpin then return end
	lastWheelA, lastWheelSpin = a, wheelSpin
	local sp = CFrame.Angles(wheelSpin, 0, 0)
	for _, wh in ipairs(wheels) do
		local base = wh.out:Lerp(wh.inn, a)
		for _, e in ipairs(wh.parts) do
			if e.w and e.w.Parent then
				if e.spin then e.w.C0 = base * sp * e.off else e.w.C0 = base * e.off end
			end
		end
	end
end

			local ra = rOf(Y0)
			local closed, len = slabCF(th, Y0, ra, Y2, R2)

			addDoor("DoorOuter", BLOCK, Vector3.new(wallW(ra), len, 0.28), closed, C.hull, MAT.Metal, { collide = true })
			addDoor("DoorLowerTrim", BLOCK, Vector3.new(wallW(ra) + 0.04, 0.24, 0.34),
				closed * CFrame.new(0, -len / 2 + 0.14, 0), C.hull2, MAT.Metal)
			-- Внешняя ручка двери, чтобы удобно нажимать снаружи
			addDoor("DoorOuterHandle", BLOCK, Vector3.new(0.95, 0.18, 0.22),
				closed * CFrame.new(0, len * 0.22, -0.22), C.dark, MAT.Metal)

			local deckRel = CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.15, -(R2 + DR) / 2)
			addDoor("DoorDeck", BLOCK, Vector3.new(wallW(R2), 0.3, R2 - DR), deckRel, C.hull, MAT.Metal, { collide = true })
			addDoor("DoorDomeRim", BLOCK, Vector3.new(wallW(DR + 0.05), 0.34, 0.45),
				CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.02, -(DR + 0.05)), C.dark, MAT.Metal, { collide = true })

			-- БОКОВОЕ СТЕКЛО КУПОЛА НА ДВЕРИ: физическое (collide = true) когда закрыто, открывается вместе с дверью!
			for i = 0, 2 do
				local a, b = rings[i], rings[i + 1]
				local pcf, plen = slabCF(th, a.y, a.r, b.y, b.r)
				local wMid = 2 * ((a.r + b.r) / 2) * TANH + 0.04
				addDoor("DoorDomeGlass", BLOCK, Vector3.new(wMid, plen, 0.12), pcf, C.glass, MAT.Glass, { tr = 0.62, collide = true })
				if i == 1 then
					addDoor("DoorGlassStreak1", BLOCK, Vector3.new(0.07, plen * 0.75, 0.13),
						pcf * CFrame.new(-0.25, 0, 0) * CFrame.Angles(0, 0, math.rad(18)), C.glassLine, MAT.SmoothPlastic, { tr = 0.45 })
					addDoor("DoorGlassStreak2", BLOCK, Vector3.new(0.05, plen * 0.55, 0.13),
						pcf * CFrame.new(0.18, -0.05, 0) * CFrame.Angles(0, 0, math.rad(18)), C.glassLine, MAT.SmoothPlastic, { tr = 0.5 })
				end
				if i == 2 then
					addDoor("DoorGlassTopFrame", BLOCK, Vector3.new(wMid + 0.06, 0.18, 0.16),
						pcf * CFrame.new(0, plen / 2, 0), C.hull, MAT.Metal, { collide = true })
				end
			end
		end

		local edgeThetas = (s.sx < 0) and { math.rad(68.5), math.rad(111.5) } or { math.rad(291.5), math.rad(248.5) }
		for _, eth in ipairs(edgeThetas) do
			for i = 0, 2 do
				local a, b = rings[i], rings[i + 1]
				local fcf, flen = slabCF(eth, a.y, a.r, b.y, b.r)
				addDoor("DoorGlassSideFrame", BLOCK, Vector3.new(0.18, flen + 0.05, 0.16),
					fcf, C.hull, MAT.Metal, { collide = true })
			end
		end

		-- ВНУТРЕННЯЯ ОБШИВКА, РУЧКА И КОРИЧНЕВАЯ СТЁГАНАЯ ПОДУШКА С 4 ПУГОВИЦАМИ (нажатие изнутри тоже закрывает/открывает дверь!)
		local midClosed, midLen = slabCF(s.theta, Y0, rOf(Y0), Y2, R2)
		local innerBase = midClosed * CFrame.new(0, 0, 0.22)

		addDoor("DoorInnerPanel", BLOCK, Vector3.new(3.6, midLen * 0.88, 0.22),
			innerBase, C.innerDoor, MAT.SmoothPlastic)
		addDoor("DoorInnerBorder", BLOCK, Vector3.new(3.75, midLen * 0.92, 0.14),
			innerBase * CFrame.new(0, 0, -0.05), C.dark, MAT.Metal)

		local latchCF = innerBase * CFrame.new(0, midLen * 0.30, 0.12)
		addDoor("DoorLatchBox", BLOCK, Vector3.new(3.3, 0.52, 0.18), latchCF, C.innerDoor, MAT.Metal)
		addDoor("DoorLatchSlot", BLOCK, Vector3.new(2.5, 0.20, 0.16), latchCF * CFrame.new(-0.15, 0, 0.04), C.black, MAT.SmoothPlastic)
		addDoor("DoorInnerHandle", BLOCK, Vector3.new(1.55, 0.11, 0.18), latchCF * CFrame.new(0.1, 0, 0.08), C.silver, MAT.Metal)
		addDoor("DoorInnerPivot", CYL, Vector3.new(0.20, 0.28, 0.28),
			latchCF * CFrame.new(0.85, 0, 0.08) * CFrame.Angles(0, math.pi / 2, 0), C.hull2, MAT.Metal)

		local cushCF = innerBase * CFrame.new(0, -midLen * 0.14, 0.34)
		addDoor("DoorCushionBase", BLOCK, Vector3.new(3.35, 1.38, 0.48), cushCF, C.cushion, MAT.Fabric)
		addDoor("DoorCushionPuff", BLOCK, Vector3.new(3.05, 1.15, 0.24), cushCF * CFrame.new(0, 0, 0.18), C.cushionHi, MAT.Fabric)
		for _, cy in ipairs({ -0.58, 0.58 }) do
			addDoor("DoorCushionRollH", CYL, Vector3.new(3.2, 0.38, 0.38),
				cushCF * CFrame.new(0, cy, 0.08) * CFrame.Angles(0, 0, math.pi / 2), C.cushion, MAT.Fabric)
		end
		for _, cx in ipairs({ -1.52, 1.52 }) do
			addDoor("DoorCushionRollV", CYL, Vector3.new(1.25, 0.38, 0.38),
				cushCF * CFrame.new(cx, 0, 0.08), C.cushion, MAT.Fabric)
		end
		for _, bx in ipairs({ -0.68, 0.68 }) do
			for _, by in ipairs({ -0.28, 0.28 }) do
				addDoor("DoorCushionBtn", BALL, Vector3.new(0.22, 0.22, 0.18),
					cushCF * CFrame.new(bx, by, 0.28), C.cushionBt, MAT.Fabric)
			end
		end

		-- ШИРОКИЙ ПОЛОГИЙ ТРАП: соединяет землю прямо с полом салона, чтобы заходить пешком без прыжков!
		local T, B = Vector2.new(8.2, 0.40), Vector2.new(14.5, -2.18)
		local rlen = (T - B).Magnitude
		local rang = math.atan2(T.Y - B.Y, T.X - B.X)
		local openCF = CFrame.new(s.sx * (T.X + B.X) / 2, (T.Y + B.Y) / 2, 0.35) * CFrame.Angles(0, 0, -s.sx * rang)
		local closedCF = CFrame.new(s.sx * 5.8, -0.18, 0.35)
		local function ramp(name, size, off, color, mat, solid)
			local rel = closedCF * off
			local p, w = mk(name, BLOCK, size, rel, color, mat, { query = false })
			table.insert(anim[grp], {
				w = w,
				part = solid and p or nil,
				coll = solid and function(a) return a > 0.35 end or nil,
				fn = function(a) return closedCF:Lerp(openCF, a) * off end,
			})
			return p
		end
		ramp("Ramp", Vector3.new(rlen + 0.6, 0.24, 4.6), CFrame.new(), Color3.fromRGB(70, 74, 74), MAT.DiamondPlate, true)
		ramp("RampEdge", Vector3.new(rlen + 0.4, 0.08, 0.14), CFrame.new(0, 0.14, 2.22), C.orange, MAT.Neon)
		ramp("RampEdge", Vector3.new(rlen + 0.4, 0.08, 0.14), CFrame.new(0, 0.14, -2.22), C.orange, MAT.Neon)
	end

	-- КАПОТ (спереди) и БАГАЖНИК (сзади) — открываются кликом прямо по ним
	compartment("hood", -1)
	compartment("trunk", 1)

	ship.Parent = Workspace
	setLights(state.lights)
end

function setLights(on)
	state.lights = on
	for _, l in ipairs(lensList) do
		if l and l.Parent then
			l.Material = on and MAT.Neon or MAT.SmoothPlastic
		end
	end
	for _, l in ipairs(lightList) do
		if l and l.Parent then l.Enabled = on end
	end
	for _, b in ipairs(beamConeList) do
		if b and b.Parent then b.Enabled = on end
	end
	if not on then
		for _, rl in ipairs(roadLights) do
			if rl.spot and rl.spot.Parent then rl.spot.Transparency = 1 end
		end
	end
	if refreshUI then refreshUI() end
end

local fpFolder
local fpArms = {}
local fpYaw, fpPitch = 0, 0

local function newFP(name, shape, color, mat)
	local p = Instance.new("Part")
	p.Name = name
	p.Shape = shape
	p.Size = Vector3.new(0.4, 0.4, 0.4)
	p.Color = color
	p.Material = mat or MAT.SmoothPlastic
	p.Anchored = true
	p.CanCollide = false
	p.CanTouch = false
	p.CanQuery = false
	p.Massless = true
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.Parent = fpFolder
	return p
end

local function skinColor(char)
	for _, n in ipairs({ "LeftHand", "LeftLowerArm", "Left Arm", "RightHand", "Right Arm" }) do
		local p = char:FindFirstChild(n)
		if p and p:IsA("BasePart") then return p.Color end
	end
	return Color3.fromRGB(234, 184, 146)
end

local function buildArms(char)
	if fpFolder then fpFolder:Destroy() end
	fpFolder = Instance.new("Folder")
	fpFolder.Name = "SaucerXArms"
	fpFolder.Parent = Workspace
	local skin = skinColor(char)
	local sleeve = skin
	if char:FindFirstChildOfClass("Shirt") then sleeve = Color3.fromRGB(72, 84, 104) end
	fpArms = {}
	for i, side in ipairs({ -1, 1 }) do
		fpArms[i] = {
			side = side,
			shoulder = (side < 0) and SHOULDER_L or SHOULDER_R,
			phi = (side < 0) and (math.pi - 0.2) or 0.2,
			upper = newFP("Upper", BLOCK, sleeve),
			fore = newFP("Fore", BLOCK, skin),
			elbow = newFP("Elbow", BALL, sleeve),
			shoulderB = newFP("Shoulder", BALL, sleeve),
			palm = newFP("Palm", BLOCK, skin),
			fingers = newFP("Fingers", BLOCK, skin),
			thumb = newFP("Thumb", BLOCK, skin),
		}
	end
end

local function segment(part, a, b, thick)
	local d = b - a
	local len = math.max(d.Magnitude, 0.05)
	part.Size = Vector3.new(thick, thick, len)
	part.CFrame = CFrame.lookAt(a, a + (d.Magnitude > 1e-3 and d or Vector3.new(0, 0, -1))) * CFrame.new(0, 0, -len / 2)
end

local function solveArm(shoulder, target, l1, l2, pole)
	local d = target - shoulder
	local dist = math.clamp(d.Magnitude, 0.3, (l1 + l2) * 0.995)
	local dir = d.Unit
	local a = (l1 * l1 - l2 * l2 + dist * dist) / (2 * dist)
	local h = math.sqrt(math.max(l1 * l1 - a * a, 0))
	local side = pole - dir * pole:Dot(dir)
	if side.Magnitude < 1e-3 then side = Vector3.new(0, -1, 0) end
	local elbow = shoulder + dir * a + side.Unit * h
	local wrist = shoulder + dir * dist
	return elbow, wrist
end

local function hideCharacter(char, v)
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then d.LocalTransparencyModifier = v end
	end
end

local function updateFP()
	local cam = Workspace.CurrentCamera
	local char = lp.Character
	if not (cam and char and root and fpFolder) then return end
	local base = root.CFrame
	local eyePos = (state.seatSide < 0) and EYE_L or EYE_R

	cam.CameraType = Enum.CameraType.Scriptable
	cam.FieldOfView = 80
	cam.CFrame = base * CFrame.new(eyePos) * CFrame.Angles(0, fpYaw, 0) * CFrame.Angles(fpPitch, 0, 0)
	pcall(function() UIS.MouseBehavior = Enum.MouseBehavior.LockCenter end)
	hideCharacter(char, 1)

	local W = base * WHEEL_CF
	for _, arm in ipairs(fpArms) do
		local shoulder = (base * CFrame.new(arm.shoulder)).Position
		local rim = W * CFrame.Angles(0, 0, steerAng + arm.phi) * CFrame.new(WHEEL_R, 0, 0)
		local grip = (rim * CFrame.new(0, 0, 0.26)).Position
		local pole = base:VectorToWorldSpace(Vector3.new(arm.side * 0.9, -1, 0.5))
		local elbow, wrist = solveArm(shoulder, grip, UPPER, FORE, pole)

		segment(arm.upper, shoulder, elbow, 0.52)
		segment(arm.fore, elbow, wrist, 0.42)
		arm.elbow.Size = Vector3.new(0.5, 0.5, 0.5); arm.elbow.Position = elbow
		arm.shoulderB.Size = Vector3.new(0.56, 0.56, 0.56); arm.shoulderB.Position = shoulder

		arm.palm.Size = Vector3.new(0.42, 0.34, 0.2)
		arm.palm.CFrame = rim * CFrame.new(0, 0, 0.2)
		arm.fingers.Size = Vector3.new(0.14, 0.36, 0.34)
		arm.fingers.CFrame = rim * CFrame.new(0.17, 0, 0.08)
		arm.thumb.Size = Vector3.new(0.12, 0.14, 0.26)
		arm.thumb.CFrame = rim * CFrame.new(-0.18, 0.08 * arm.side, 0.14) * CFrame.Angles(0, 0, 0.3 * arm.side)
	end
end

function setFirstPerson(on)
	if on then
		if not state.seated then
			say("Сначала зайди в корабль и сядь на кресло, затем V")
			return
		end
		local char = lp.Character
		if not char then return end
		buildArms(char)
		fpYaw, fpPitch = 0, 0
		state.fp = true
		say("1-е лицо: поворачивай мышь / палец для направления полёта, R — центр")
	else
		state.fp = false
		if fpFolder then fpFolder:Destroy() fpFolder = nil end
		fpArms = {}
		local cam = Workspace.CurrentCamera
		local char = lp.Character
		if char then hideCharacter(char, 0) end
		pcall(function() UIS.MouseBehavior = Enum.MouseBehavior.Default end)
		if cam then
			cam.CameraType = Enum.CameraType.Custom
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum then cam.CameraSubject = hum end
			cam.FieldOfView = 70
		end
	end
	if refreshUI then refreshUI() end
end

function destroyShip()
	if state.fp then setFirstPerson(false) end
	if state.seated then exitShip() end
	if ship then ship:Destroy() end
	ship, root, beacon = nil, nil, nil
	spinners, wheels, pulsers, podFlames, roadLights = {}, {}, {}, {}, {}
	seatPartsL, seatPartsR = {}, {}
	clickableParts = {}
	partActions = {}
	state.doorL, state.doorR, state.hood, state.trunk, state.gear = false, false, false, false, false
	if refreshUI then refreshUI() end
end

local function spawnShip()
	destroyShip()
	local char = lp.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then say("Персонаж не найден") return end
	local look = hrp.CFrame.LookVector
	look = Vector3.new(look.X, 0, look.Z)
	if look.Magnitude < 0.01 then look = Vector3.new(0, 0, -1) end
	look = look.Unit
	local p = hrp.Position + look * 26
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { char }
	local hit = Workspace:Raycast(p + Vector3.new(0, 60, 0), Vector3.new(0, -300, 0), rp)
	local y = hit and (hit.Position.Y + 2.1) or (hrp.Position.Y - 1.0)
	yaw = math.atan2(-look.X, -look.Z) + math.pi
	pitch, roll, vel = 0, 0, Vector3.zero
	pos = Vector3.new(p.X, y, p.Z)
	steerAng, wheelSpin = 0, 0
	build(CFrame.new(pos) * CFrame.Angles(0, yaw, 0))
	refreshUI()
	say("Нажми прямо на дверь, зайди по трапу в салон и сядь на кресло!")
end

-- Самостоятельная посадка в кресло и выход в салон пешком
---------------------------------------------------------------------
function sitDown(hum, hrp, side)
	if state.seated or not root then return end
	yaw = select(2, root.CFrame:ToEulerAnglesYXZ())
	pitch, roll, vel = 0, 0, Vector3.zero
	pos = root.Position
	state.seated = true
	state.seatSide = side or -1
	hum.AutoRotate = false
	hrp.Anchored = true
	hrp.AssemblyLinearVelocity = Vector3.zero
	hrp.CFrame = root.CFrame * ((state.seatSide < 0) and seatRelL or seatRelR)
	hum.Sit = true
	refreshUI()
	say("За рулём! Смотри камерой вверх и жми вперёд для взлёта. Прыжок / G — встать с кресла")
end

function exitShip()
	if not state.seated then return end
	lastExitTime = os.clock()
	if state.fp then setFirstPerson(false) end
	state.seated = false
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp and root then
		hrp.Anchored = false
		-- Ставим игрока прямо на пол внутри салона рядом с креслом, чтобы он мог сам выйти пешком через дверь!
		hrp.CFrame = root.CFrame * CFrame.new(0, 2.95, 0.1)
		hrp.AssemblyLinearVelocity = Vector3.zero
	elseif hrp then
		hrp.Anchored = false
	end
	if hum then
		hum.Sit = false
		hum.AutoRotate = true
		hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
	refreshUI()
	say("Ты встал с кресла в салоне! Нажми на дверь, чтобы открыть её и выйти по трапу")
end

local controls
pcall(function()
	local ps = lp:FindFirstChild("PlayerScripts")
	local pm = ps and ps:FindFirstChild("PlayerModule")
	if pm then controls = require(pm):GetControls() end
end)

local function getMove()
	local mv = Vector3.zero
	if controls then
		local ok, v = pcall(function() return controls:GetMoveVector() end)
		if ok and v then mv = v end
	end
	if mv.Magnitude < 0.05 then
		local x, z = 0, 0
		if UIS:IsKeyDown(Enum.KeyCode.W) then z = z - 1 end
		if UIS:IsKeyDown(Enum.KeyCode.S) then z = z + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.D) then x = x + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.A) then x = x - 1 end
		mv = Vector3.new(x, 0, z)
	end
	return mv
end

-- Обновление динамического света фар на дороге
local function updateRoadHeadlights(char)
	if not state.lights then return end
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { ship, char, fpFolder }
	for _, rl in ipairs(roadLights) do
		if rl.holder and rl.holder.Parent and rl.spot and rl.spot.Parent then
			local origin = rl.holder.Position
			local dir = rl.holder.CFrame.LookVector * 85
			local hit = Workspace:Raycast(origin, dir, rp)
			if hit then
				local dist = (hit.Position - origin).Magnitude
				local diam = math.clamp(4.2 + dist * 0.26, 4.5, 18)
				rl.spot.Size = Vector3.new(0.08, diam, diam)
				rl.spot.CFrame = CFrame.lookAt(hit.Position + hit.Normal * 0.06, hit.Position + hit.Normal * 2) * CFrame.Angles(0, math.pi / 2, 0)
				rl.spot.Transparency = math.clamp(0.76 + (dist / 85) * 0.20, 0.76, 0.96)
				if rl.light then
					rl.light.Brightness = math.clamp(3.2 * (1 - dist / 95), 0.8, 3.2)
					rl.light.Range = math.clamp(diam * 1.4, 12, 28)
				end
			else
				rl.spot.Transparency = 1
			end
		end
	end
end

table.insert(connections, RunService.RenderStepped:Connect(function(dt)
	if not (ship and ship.Parent and root) then return end
	local t = os.clock()
	if beacon then beacon.Transparency = (math.sin(t * 5) > 0) and 0 or 0.7 end

	fanAng = fanAng + dt * (state.hood and 16 or 3)
	updateSpinners()
	updatePulses(t)
	updateWheels()

	local char = lp.Character
	updateRoadHeadlights(char)

	if not state.seated then
		updatePodFlames(t, dt, state.forceFire, 1, state.forceFire and 0.8 or 0)
		if char and (t - lastExitTime > 1.8) then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hum and hrp and hum.Health > 0 then
				local pL = (root.CFrame * seatRelL).Position
				local pR = (root.CFrame * seatRelR).Position
				if (hrp.Position - pL).Magnitude < 2.05 then
					sitDown(hum, hrp, -1)
					return
				elseif (hrp.Position - pR).Magnitude < 2.05 then
					sitDown(hum, hrp, 1)
					return
				end
			end
		end
		return
	end

	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) or hum.Health <= 0 then exitShip() return end

	local mv = getMove()
	local fwd = -mv.Z
	local strafe = mv.X
	local boost = UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 1.85 or 1

	-- Определяем направление взгляда камеры для полноценного 3D-полёта!
	local cam = Workspace.CurrentCamera
	local camLook = cam and cam.CFrame.LookVector or CFrame.Angles(0, yaw, 0).LookVector
	if state.fp then
		camLook = (CFrame.Angles(0, yaw + fpYaw, 0) * CFrame.Angles(fpPitch, 0, 0)).LookVector
	end

	-- Дополнительная тонкая коррекция высоты с клавиатуры (E вверх, Q/Ctrl вниз) по желанию
	local manualVert = 0
	if UIS:IsKeyDown(Enum.KeyCode.E) then manualVert = manualVert + 1 end
	if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.Q) then manualVert = manualVert - 1 end

	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { ship, char, fpFolder }
	local hit = Workspace:Raycast(pos + Vector3.new(0, 3, 0), Vector3.new(0, -600, 0), rp)
	local groundY = hit and hit.Position.Y or -1e5
	local clearance = hit and (pos.Y - 2.1 - groundY) or 999

	-- Вертикальная тяга от угла наклона камеры: смотришь вверх и жмёшь вперёд -> взлетаешь в небо!
	local climbSignal = camLook.Y * fwd + manualVert * 0.75
	local wantsTakeoff = (climbSignal > 0.14) or (manualVert > 0)
	local onGround = (hit ~= nil) and (clearance < 0.65) and (not wantsTakeoff)

	if onGround then
		-- Езда на колёсах по земле
		local look2D = CFrame.Angles(0, yaw, 0).LookVector
		local fs = vel:Dot(look2D)
		local gain = math.clamp(math.abs(fs) / 10, 0.25, 1)
		yaw = yaw - strafe * dt * 1.9 * gain * ((fs < -0.5) and -1 or 1)
		local heading = CFrame.Angles(0, yaw, 0)
		local target = heading.LookVector * (fwd * speed * boost)
		vel = vel:Lerp(target, math.clamp(dt * 3.2, 0, 1))
	else
		-- НАСТОЯЩИЙ СВОБОДНЫЙ 3D-ПОЛЁТ:
		-- Корабль плавно поворачивается по курсу камеры при движении и летит точно туда, куда ты смотришь (вверх, вниз, виражи)!
		local flatCam = Vector3.new(camLook.X, 0, camLook.Z)
		if flatCam.Magnitude > 0.05 and (math.abs(fwd) > 0.05 or math.abs(strafe) > 0.05) then
			local targetYaw = math.atan2(-flatCam.X, -flatCam.Z)
			local diff = math.atan2(math.sin(targetYaw - yaw), math.cos(targetYaw - yaw))
			yaw = yaw + diff * math.min(dt * 3.6, 1)
		end
		yaw = yaw - strafe * dt * 1.65

		local heading = CFrame.Angles(0, yaw, 0)
		local flyPitch = math.asin(math.clamp(camLook.Y, -0.88, 0.88))
		local flyDir = (heading * CFrame.Angles(flyPitch, 0, 0)).LookVector
		local rightDir = heading.RightVector

		local target = flyDir * (fwd * speed * boost)
			+ rightDir * (strafe * speed * 0.55)
			+ Vector3.new(0, manualVert * speed * 0.65, 0)

		vel = vel:Lerp(target, math.clamp(dt * 2.8, 0, 1))
	end

	pos = pos + vel * dt

	if hit then
		local floorY = groundY + 2.1
		if onGround then
			pos = Vector3.new(pos.X, pos.Y + (floorY - pos.Y) * math.min(dt * 10, 1), pos.Z)
			vel = Vector3.new(vel.X, 0, vel.Z)
		elseif pos.Y < floorY then
			pos = Vector3.new(pos.X, floorY, pos.Z)
			if vel.Y < 0 then vel = Vector3.new(vel.X, 0, vel.Z) end
		end
	end

	-- Умные колёса: сами убираются в полёте и выдвигаются при посадке
	local wantUp = state.gear
	if not hit or clearance > 6.5 then
		wantUp = true
	elseif clearance < 3.5 + math.max(0, -vel.Y) * 0.6 then
		wantUp = false
	end
	if wantUp ~= state.gear then setGroup("gear", wantUp) end

	local heading = CFrame.Angles(0, yaw, 0)
	local look = heading.LookVector
	if onGround then wheelSpin = wheelSpin - vel:Dot(look) / 1.1 * dt end

	-- СИНИЙ ОГОНЬ ИЗ КАПСУЛ В ПОЛЁТЕ
	local isFlying = (not onGround) or (vel.Magnitude > 3 and not onGround) or (math.abs(fwd) > 0.1 and boost > 1) or state.forceFire
	local spdRatio = math.clamp(vel.Magnitude / math.max(speed, 1), 0, 1.5)
	updatePodFlames(t, dt, isFlying, boost, spdRatio)

	steerAng = steerAng + (-strafe * 1.3 - steerAng) * math.min(dt * 8, 1)

	-- Реалистичный наклон корпуса (тангаж по вертикальной скорости и крен при виражах)
	local climbPitch = math.clamp(vel.Y / math.max(speed, 1), -0.65, 0.65) * 0.48
	local movePitch  = -math.clamp(fwd, -1, 1) * 0.10
	local tp = onGround and 0 or (climbPitch + movePitch)
	local tr = -math.clamp(strafe, -1, 1) * (onGround and 0.05 or 0.36)
	local k = math.min(dt * 4.5, 1)
	pitch = pitch + (tp - pitch) * k
	roll = roll + (tr - roll) * k

	local bob = onGround and 0 or (math.sin(t * 2.2) * 0.10)
	root.CFrame = CFrame.new(pos + Vector3.new(0, bob, 0)) * heading * CFrame.Angles(pitch, 0, roll)

	hrp.CFrame = root.CFrame * ((state.seatSide < 0) and seatRelL or seatRelR)
	hrp.AssemblyLinearVelocity = Vector3.zero
	if not hum.Sit then hum.Sit = true end

	if state.fp then updateFP() end
end))

local function corner(o, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r)
	c.Parent = o
	return c
end

local function tile(parent, order, icon, label, hint, cb)
	local b = Instance.new("TextButton")
	b.Name = label
	b.LayoutOrder = order
	b.AutoButtonColor = false
	b.Text = ""
	b.BackgroundColor3 = COL.tile
	b.BorderSizePixel = 0
	b.Parent = parent
	corner(b, 8)

	local ic = Instance.new("TextLabel")
	ic.BackgroundTransparency = 1
	ic.Size = UDim2.new(1, 0, 0, 20)
	ic.Position = UDim2.new(0, 0, 0, 3)
	ic.Font = Enum.Font.GothamBold
	ic.TextSize = 16
	ic.Text = icon
	ic.TextColor3 = COL.text
	ic.Parent = b

	local lb = Instance.new("TextLabel")
	lb.BackgroundTransparency = 1
	lb.Size = UDim2.new(1, 0, 0, 12)
	lb.Position = UDim2.new(0, 0, 1, -15)
	lb.Font = Enum.Font.GothamMedium
	lb.TextSize = 9
	lb.Text = label
	lb.TextColor3 = COL.sub
	lb.Parent = b

	local t = { btn = b, icon = ic, label = lb, on = false }
	local function paint(hover)
		local bg = t.on and COL.on or (hover and COL.hover or COL.tile)
		TweenService:Create(b, TweenInfo.new(0.12), { BackgroundColor3 = bg }):Play()
		lb.TextColor3 = t.on and COL.darkTx or COL.sub
	end
	t.set = function(on)
		t.on = on and true or false
		paint(false)
	end
	b.MouseEnter:Connect(function()
		paint(true)
		if statusLabel then statusLabel.Text = hint end
	end)
	b.MouseLeave:Connect(function()
		paint(false)
		if statusLabel then statusLabel.Text = lastMsg end
	end)
	b.MouseButton1Click:Connect(cb)
	return t
end

	if ui.fire then ui.fire.set(state.forceFire) end
end

-- Дополнительный обработчик нажатия пальцем / мышью по 3D-деталям (двери, капот, багажник, кресла)
local function raycastInteract(screenPos, isViewport)
	if not (ship and ship.Parent and #clickableParts > 0) then return end
	local cam = Workspace.CurrentCamera
	if not cam then return end
	local ray = isViewport and cam:ViewportPointToRay(screenPos.X, screenPos.Y) or cam:ScreenPointToRay(screenPos.X, screenPos.Y)
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Include
	rp.FilterDescendantsInstances = clickableParts
	local hit = Workspace:Raycast(ray.Origin, ray.Direction * 60, rp)
	if hit and hit.Instance then
		local char = lp.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if hrp and (hrp.Position - hit.Position).Magnitude > 44 then return end
		local act = partActions[hit.Instance]
		if act then triggerAction(act) end
	end
end

table.insert(connections, UIS.TouchTapInWorld:Connect(function(pos2D, processedByUI)
	if not processedByUI then raycastInteract(pos2D, false) end
end))

table.insert(connections, UIS.InputBegan:Connect(function(input, gp)
	if gp then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		local mPos = UIS:GetMouseLocation()
		raycastInteract(mPos, true)
	end
	local k = input.KeyCode
	if k == Enum.KeyCode.G then
		if state.seated then
			exitShip()
		else
			local char = lp.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hum and hrp and root and (hrp.Position - (root.CFrame * seatRelL).Position).Magnitude <= 8 then
				sitDown(hum, hrp, -1)
			else
				say("Нажми на дверь, зайди внутрь по трапу и подойди к креслу!")
			end
		end
	elseif k == Enum.KeyCode.L then setLights(not state.lights)
	elseif k == Enum.KeyCode.V then setFirstPerson(not state.fp)
	elseif k == Enum.KeyCode.R then fpYaw, fpPitch = 0, 0
	elseif k == Enum.KeyCode.B and ship and not state.seated then setGroup("gear", not state.gear)
	elseif k == Enum.KeyCode.F then
		state.forceFire = not state.forceFire
		refreshUI()
	end
end))

-- Осмотр и управление направлением в режиме от 1-го лица (мышь и палец)
table.insert(connections, UIS.InputChanged:Connect(function(input, gp)
	if not (state.seated and state.fp) then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement then
		fpYaw = math.clamp(fpYaw - input.Delta.X * 0.004, -2.4, 2.4)
		fpPitch = math.clamp(fpPitch - input.Delta.Y * 0.004, -1.2, 1.2)
	elseif input.UserInputType == Enum.UserInputType.Touch and not gp then
		fpYaw = math.clamp(fpYaw - input.Delta.X * 0.007, -2.4, 2.4)
		fpPitch = math.clamp(fpPitch - input.Delta.Y * 0.007, -1.2, 1.2)
	end
end))

env.__SaucerX = function()
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	pcall(function() if state.fp then setFirstPerson(false) end end)
	pcall(destroyShip)
	if gui then gui:Destroy() end
end
